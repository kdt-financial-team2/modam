CREATE USER modam IDENTIFIED BY 1234;

GRANT DBA TO modam;

-- 1. MEMBER 테이블 (회원)
CREATE TABLE member (
    id NUMBER PRIMARY KEY,
    name VARCHAR2(50) NOT NULL,
    user_id VARCHAR2(50) NOT NULL UNIQUE,
    pw_hash VARCHAR2(255) NOT NULL, -- BCrypt
    email VARCHAR2(100) NOT NULL UNIQUE,
    agree CHAR(1) DEFAULT 'N' NOT NULL,
    notif CHAR(1) DEFAULT 'N' NOT NULL,
    en_first VARCHAR2(50) NOT NULL,
    en_last VARCHAR2(50) NOT NULL,
    bank_name VARCHAR2(50) NOT NULL,
    pers_acct_no VARCHAR2(255) NOT NULL, -- AES-256 암호화 대비
    rrn VARCHAR2(255) NOT NULL, -- AES-256 암호화 대비
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    active CHAR(1) DEFAULT 'Y' NOT NULL,
    address VARCHAR2(255) NOT NULL,
    role VARCHAR2(20) DEFAULT 'USER' NOT NULL,
    phone_no VARCHAR2(20) NOT NULL,
    profile_img VARCHAR2(500) -- 개인 프로필 이미지 추가
);

-- 2. ACCOUNT 테이블 (계좌)
CREATE TABLE account (
    id NUMBER PRIMARY KEY,
    acct_no VARCHAR2(30) NOT NULL UNIQUE,
    pw_hash VARCHAR2(255),
    status VARCHAR2(20) DEFAULT 'ACTIVE' NOT NULL,
    bal NUMBER DEFAULT 0 NOT NULL,
    avail_bal NUMBER DEFAULT 0 NOT NULL,
    limit_amt NUMBER,
    deliv_addr VARCHAR2(255),
    job_info VARCHAR2(100),
    trade_purp VARCHAR2(100),
    fund_src VARCHAR2(100),
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP
);

-- 3. ACCOUNT_MEMBER 테이블 (계좌-회원 매핑)
CREATE TABLE account_member (
    id NUMBER PRIMARY KEY,
    acc_id NUMBER NOT NULL,
    mem_id NUMBER NOT NULL,
    agree_cl CHAR(1) DEFAULT 'N' NOT NULL,
    inv_status VARCHAR2(20) DEFAULT 'WAIT' NOT NULL,
    tot_depo NUMBER DEFAULT 0 NOT NULL,
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_am_acc FOREIGN KEY (acc_id) REFERENCES account(id),
    CONSTRAINT fk_am_mem FOREIGN KEY (mem_id) REFERENCES member(id)
);

-- 4. CARD 테이블 (카드)
CREATE TABLE card (
    id NUMBER PRIMARY KEY,
    acc_id NUMBER NOT NULL,
    mem_id NUMBER NOT NULL,
    card_no VARCHAR2(255) NOT NULL UNIQUE, -- AES-256 암호화 대비
    exp_date VARCHAR2(5) NOT NULL,
    status VARCHAR2(20) DEFAULT 'ACTIVE' NOT NULL,
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_card_acc FOREIGN KEY (acc_id) REFERENCES account(id),
    CONSTRAINT fk_card_mem FOREIGN KEY (mem_id) REFERENCES member(id)
);

-- 5. TRANSACTION 테이블 (금액 기록)
CREATE TABLE transaction (
    id NUMBER PRIMARY KEY,
    acc_id NUMBER NOT NULL,
    card_id NUMBER,
    mem_id NUMBER NOT NULL,
    tx_type VARCHAR2(20) NOT NULL,
    category VARCHAR2(50),
    amt NUMBER NOT NULL,
    aft_bal NUMBER NOT NULL,
    merch_nm VARCHAR2(100),
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT fk_tx_acc FOREIGN KEY (acc_id) REFERENCES account(id),
    CONSTRAINT fk_tx_card FOREIGN KEY (card_id) REFERENCES card(id),
    CONSTRAINT fk_tx_mem FOREIGN KEY (mem_id) REFERENCES member(id)
);

-- 6. SPEND_RECORD 테이블 (이미지 및 메모)
CREATE TABLE spend_record (
    id NUMBER PRIMARY KEY,
    tx_id NUMBER NOT NULL UNIQUE,
    img_url VARCHAR2(500),
    memo VARCHAR2(1000),
    emoticon VARCHAR2(50),
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_sr_tx FOREIGN KEY (tx_id) REFERENCES transaction(id)
);

-- 7. COMMENT 테이블 (댓글)
CREATE TABLE "comment" (
    id NUMBER PRIMARY KEY,
    record_id NUMBER NOT NULL,
    mem_id NUMBER NOT NULL,
    content VARCHAR2(500) NOT NULL,
    emoticon VARCHAR2(50),
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_com_rec FOREIGN KEY (record_id) REFERENCES spend_record(id),
    CONSTRAINT fk_com_mem FOREIGN KEY (mem_id) REFERENCES member(id)
);

-- 8. SAVINGS 테이블 (저축)
CREATE TABLE savings (
    id NUMBER PRIMARY KEY,
    acct_id NUMBER NOT NULL,
    save_type VARCHAR2(50) NOT NULL,
    target_amt NUMBER NOT NULL,
    curr_amt NUMBER DEFAULT 0 NOT NULL,
    target_dt DATE,
    is_auto CHAR(1) DEFAULT 'N' NOT NULL,
    auto_amt NUMBER,
    auto_cycle VARCHAR2(20),
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_sav_acc FOREIGN KEY (acct_id) REFERENCES account(id)
);

-- 9. COUPLE 테이블 (커플 프로필)
CREATE TABLE couple (
    id NUMBER PRIMARY KEY,
    acct_id NUMBER NOT NULL UNIQUE,
    inv_code VARCHAR2(50) NOT NULL,
    d_day DATE,
    acct_alias VARCHAR2(100),
    couple_img VARCHAR2(500), -- 커플 공동 이미지 추가
    inv_email VARCHAR2(100),
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_cp_acc FOREIGN KEY (acct_id) REFERENCES account(id)
);

-- 10. POINT_HISTORY 테이블 (포인트 내역 — 공동 계좌 기준)
CREATE TABLE point_history (
    id NUMBER PRIMARY KEY,
    couple_id NUMBER NOT NULL,
    type VARCHAR2(10) NOT NULL,
    reason VARCHAR2(30) NOT NULL,
    amt NUMBER NOT NULL,
    aft_bal NUMBER NOT NULL,
    descrip VARCHAR2(255) NOT NULL,
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT fk_ph_couple FOREIGN KEY (couple_id) REFERENCES couple(id)
);

-- 11. ITEM 테이블 (상점 아이템)
CREATE TABLE item (
    id NUMBER PRIMARY KEY,
    item_name VARCHAR2(100) NOT NULL,
    item_type VARCHAR2(20) NOT NULL,
    price NUMBER NOT NULL,
    img_url VARCHAR2(500),
    is_active CHAR(1) DEFAULT 'Y' NOT NULL,
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP
);

-- 12. INVENTORY 테이블 (아이템 보유함)
CREATE TABLE inventory (
    id NUMBER PRIMARY KEY,
    mem_id NUMBER NOT NULL,
    item_id NUMBER NOT NULL,
    is_applied CHAR(1) DEFAULT 'N' NOT NULL,
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_inv_mem FOREIGN KEY (mem_id) REFERENCES member(id),
    CONSTRAINT fk_inv_item FOREIGN KEY (item_id) REFERENCES item(id),
    CONSTRAINT uk_mem_item UNIQUE (mem_id, item_id)
);

-- 13. NOTIFICATION 테이블 (알림)
CREATE TABLE notification (
    id NUMBER PRIMARY KEY,
    mem_id NUMBER NOT NULL,
    noti_type VARCHAR2(20) NOT NULL,
    message VARCHAR2(500) NOT NULL,
    target_url VARCHAR2(255),
    is_read CHAR(1) DEFAULT 'N' NOT NULL,
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    CONSTRAINT fk_noti_mem FOREIGN KEY (mem_id) REFERENCES member(id)
);
